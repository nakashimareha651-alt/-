        let currentQuiz = null; // 現在出題中のクイズを保持する変数

        // ランダムでクイズを出力する関数
        function generateQuiz() {
            // 答えを隠し、リセット
            document.getElementById('quiz-answer').style.display = 'none';
            document.getElementById('quiz-answer').textContent = '';

            // ランダムにクイズを選ぶ
            const randomIndex = Math.floor(Math.random() * quizData.length);
            currentQuiz = quizData[randomIndex];

            // クイズを出力
            const outputElement = document.getElementById('quiz-output');
            outputElement.innerHTML = `<strong>【${currentQuiz.category}に関する問題！】</strong><br>${currentQuiz.question}`;
        }

        // 答えを表示する関数
        function showAnswer() {
            const answerElement = document.getElementById('quiz-answer');

            if (currentQuiz) {
                answerElement.textContent = `正解は...「${currentQuiz.key}」でした！`;
                answerElement.style.display = 'block';
            } else {
                answerElement.textContent = 'まずは「次のクイズを出す！」ボタンを押してね！';
                answerElement.style.display = 'block';
            }
        }

        // ページ読み込み時に最初のクイズを出す
        window.onload = generateQuiz;
